cmake_minimum_required(VERSION 3.22)
project(RemarkableInfiniteHorizontal VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)

option(INFINITE_HORIZONTAL_DEVICE_BUILD "Use the matched Paper Pure SDK" OFF)
if(INFINITE_HORIZONTAL_DEVICE_BUILD)
    # The public SDK omits host copies of target-only dynamic plugins.
    set(QT_SKIP_AUTO_PLUGIN_INCLUSION ON)
    set(QT_SKIP_AUTO_QML_PLUGIN_INCLUSION ON)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Qml)
if(INFINITE_HORIZONTAL_DEVICE_BUILD)
    message(STATUS "Device compiler: ${CMAKE_CXX_COMPILER}")
    message(STATUS "Device sysroot: $ENV{SDKTARGETSYSROOT}")
    if(NOT CMAKE_CXX_COMPILER MATCHES "aarch64-remarkable-linux-g\\+\\+$"
       OR NOT "$ENV{SDKTARGETSYSROOT}" MATCHES "/cortexa55-remarkable-linux$")
        message(FATAL_ERROR "Device builds require the matched AArch64 Paper Pure SDK.")
    endif()
    if(NOT Qt6_VERSION VERSION_EQUAL "6.10.3")
        message(FATAL_ERROR "The verified SDK supplies Qt 6.10.3.")
    endif()
endif()

add_library(stock_activity src/platform/StockActivity.cpp src/platform/StockActivity.h)
target_include_directories(stock_activity PUBLIC src)
target_link_libraries(stock_activity PUBLIC Qt6::Core)

add_executable(infinite-idle src/platform/StockIdleMain.cpp)
target_link_libraries(infinite-idle PRIVATE stock_activity)
add_library(native-inspector SHARED src/ResourceInspector.cpp)
target_link_libraries(native-inspector PRIVATE Qt6::Core)
add_executable(native-document-snapshot src/DocumentSnapshot.cpp)
target_link_libraries(native-document-snapshot PRIVATE Qt6::Core)
add_library(canvas_settings src/CanvasSettings.cpp src/CanvasSettings.h)
set_target_properties(canvas_settings PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(canvas_settings PUBLIC src)
target_link_libraries(canvas_settings PUBLIC Qt6::Core)
add_library(infinite-settings SHARED src/SettingsRegistration.cpp)
target_link_libraries(infinite-settings PRIVATE canvas_settings Qt6::Qml)
add_executable(infinite-verify src/VerifyRuntime.cpp)
target_link_libraries(infinite-verify PRIVATE Qt6::Core)

foreach(target stock_activity infinite-idle native-inspector native-document-snapshot
               canvas_settings infinite-settings infinite-verify)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    add_executable(stock_activity_tests tests/StockActivityTests.cpp)
    target_link_libraries(stock_activity_tests PRIVATE stock_activity Qt6::Test)
    add_test(NAME stock_activity_tests COMMAND stock_activity_tests)
    add_executable(settings_tests tests/SettingsTests.cpp)
    target_link_libraries(settings_tests PRIVATE canvas_settings Qt6::Test)
    add_test(NAME settings_tests COMMAND settings_tests)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME manager_tests COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_manager.py")
        add_test(NAME boot_control_tests COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_boot_control.py")
        add_test(NAME runtime_verifier_tests COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_runtime_test.py"
            "$<TARGET_FILE:infinite-verify>")
    endif()
endif()
